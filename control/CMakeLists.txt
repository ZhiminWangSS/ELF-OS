cmake_minimum_required(VERSION 3.16)
project(elf_go2_control LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(UNITREE_SDK2_ROOT "$ENV{HOME}/unitree_sdk2-2.0.2" CACHE PATH "Official Unitree SDK2 checkout")
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory("${UNITREE_SDK2_ROOT}" "${CMAKE_BINARY_DIR}/unitree_sdk2" EXCLUDE_FROM_ALL)
add_executable(elf_go2 cpp/elf_go2.cpp)
target_link_libraries(elf_go2 PRIVATE unitree_sdk2)
include(CTest)
if(BUILD_TESTING)
  add_executable(guard_test tests/guard_test.cpp)
  target_include_directories(guard_test PRIVATE cpp)
  add_test(NAME motion_guard COMMAND guard_test)
endif()
