cmake_minimum_required(VERSION 3.16)
project(elf_navigation_sdk LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(UNITREE_SDK2_ROOT "$ENV{HOME}/unitree_sdk2-2.0.2" CACHE PATH "Official SDK")
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory("${UNITREE_SDK2_ROOT}" "${CMAKE_BINARY_DIR}/unitree_sdk2" EXCLUDE_FROM_ALL)
add_executable(nav_worker nav_worker.cpp)
target_link_libraries(nav_worker PRIVATE unitree_sdk2)
include(CTest)
if(BUILD_TESTING)
  add_executable(nav_policy_test policy_test.cpp)
  add_test(NAME navigation_policy COMMAND nav_policy_test)
endif()
add_executable(remote_probe remote_probe.cpp)
target_link_libraries(remote_probe PRIVATE unitree_sdk2)
